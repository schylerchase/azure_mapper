1. Create a new function `_deferComplianceUpdate` in app-core.js near the existing `_runComplianceWithCache` (around line 340):

```javascript
function _deferComplianceUpdate(ctx, sb2) {
  var run = function() {
    try {
      var findings = _runComplianceWithCache(ctx);
      _complianceFindings = findings;
      window._complianceFindings = findings;
      if (findings.length) addComplianceChip(sb2, findings);
      _addBUDRChip(sb2);
    } catch(e) { console.warn('Compliance check error:', e); }
  };
  // Cache hit path is fast — run synchronously
  var fp = _dataFingerprint();
  if (fp === _complianceDataFP && _complianceCachedFindings) {
    run();
    return;
  }
  // Cache miss — defer to idle callback
  if (typeof requestIdleCallback === 'function') {
    requestIdleCallback(run, { timeout: 2000 });
  } else {
    setTimeout(run, 50); // Safari fallback
  }
}
```

Key design choice: if the fingerprint cache hits, run synchronously (it's fast — just returns cached array). Only defer on cache miss (the expensive path that runs all framework checks). This avoids the "empty badge" problem for re-renders of the same data.
